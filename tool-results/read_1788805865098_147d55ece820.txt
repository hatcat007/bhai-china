    49→    49→function readJsonMap<T>(key: string): Record<string, T> {
    50→    50→  if (typeof window === "undefined") return {};
    51→    51→  try {
    52→    52→    const raw = window.localStorage.getItem(key);
    53→    53→    if (!raw) return {};
    54→    54→    const parsed: unknown = JSON.parse(raw);
    55→    55→    return parsed && typeof parsed === "object"
    56→    56→      ? (parsed as Record<string, T>)
    57→    57→      : {};
    58→    58→  } catch {
    59→    59→    return {};
    60→    60→  }
    61→    61→}
    62→    62→
    63→    63→function writeJsonMap(key: string, map: Record<string, unknown>): void {
    64→    64→  try {
    65→    65→    window.localStorage.setItem(key, JSON.stringify(map));
    66→    66→  } catch {
    67→    67→    /* 私密模式等场景静默失败 */
    68→    68→  }
    69→    69→}
    70→    70→
    71→    71→function readHistory(slug: string): HistoryEntry[] {
    72→    72→  const map = readJsonMap<HistoryEntry[]>(HISTORY_KEY);
    73→    73→  const list = map[slug];
    74→    74→  return Array.isArray(list) ? list : [];
    75→    75→}
    76→    76→
    77→    77→function appendHistory(slug: string, entry: HistoryEntry): HistoryEntry[] {
    78→    78→  const map = readJsonMap<HistoryEntry[]>(HISTORY_KEY);
    79→    79→  const list = [entry, ...(map[slug] ?? [])].slice(0, HISTORY_CAP);
    80→    80→  map[slug] = list;
    81→    81→  writeJsonMap(HISTORY_KEY, map);
    82→    82→  return list;
    83→    83→}
    84→    84→
    85→    85→function clearHistory(slug: string): HistoryEntry[] {
    86→    86→  const map = readJsonMap<HistoryEntry[]>(HISTORY_KEY);
    87→    87→  delete map[slug];
    88→    88→  writeJsonMap(HISTORY_KEY, map);
    89→    89→  return [];
    90→    90→}

    92→    92→function fmtTime(ts: number): string {
    93→    93→  const d = new Date(ts);
    94→    94→  const p = (n: number) => String(n).padStart(2, "0");
    95→    95→  return `${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}`;
    96→    96→}

    98→    98→function normalizeUrl(raw: string): string {
    99→    99→  const s = raw.trim();
   100→   100→  if (!s) return s;
   101→   101→  return /^https?:\/\//i.test(s) ? s : `https://${s}`;
   102→   102→}
   103→   103→
   104→   104→function isValidUrl(raw: string): boolean {
   105→   105→  try {
   106→   106→    const u = new URL(raw);
   107→   107→    return (u.protocol === "http:" || u.protocol === "https:") && !!u.hostname;
   108→   108→  } catch {
   109→   109→    return false;
   110→   110→  }
   111→   111→}

   113→   113→export function ToolRunner({ tool }: { tool: AITool }) {
   114→   114→  const { toast } = useToast();
   115→   115→  const { t } = useLang();
   116→   116→  const r = t.runner;
   117→   117→  // 展示字段（name/label/placeholder/cta/outputHint）随 locale 切换；逻辑字段仍用 tool
   118→   118→  const lt = useLocalizedTool(tool);
   119→   119→  const Icon = TOOL_ICONS[tool.icon] ?? TOOL_ICONS.ScanSearch;
   120→   120→
   121→   121→  const [values, setValues] = useState<Record<string, string>>(() =>
   122→   122→    Object.fromEntries(tool.fields.map((f) => [f.id, ""]))
   123→   123→  );
   124→   124→  const [usage, setUsage] = useState<UsageMap | null>(null); // null = localStorage 未读
   125→   125→  const [unlocked, setUnlocked] = useState(false);
   126→   126→  const [output, setOutput] = useState<string | null>(null);
   127→   127→  const [streaming, setStreaming] = useState(false);
   128→   128→  const [error, setError] = useState<string | null>(null);
   129→   129→  const [inlineError, setInlineError] = useState<string | null>(null);
   130→   130→  const [copied, setCopied] = useState(false);
   131→   131→
   132→   132→  // 扫描监视器状态（仅分析器使用）：阶段日志 / 进度 / 轮次号（重挂载用）
   133→   133→  const [stageLines, setStageLines] = useState<string[]>([]);
   134→   134→  const [progress, setProgress] = useState(0);
   135→   135→  const [runId, setRunId] = useState(0);
   136→   136→  // 提交时冻结的目标 URL：避免输入过程中实时触发截图预加载/回退链空转
   137→   137→  const [scanUrl, setScanUrl] = useState<string | null>(null);
   138→   138→  // 本工具最近生成记录（localStorage，最多 5 条）
   139→   139→  const [history, setHistory] = useState<HistoryEntry[]>([]);
   140→   140→  const [historyOpen, setHistoryOpen] = useState(false);
   141→   141→
   142→   142→  // 解锁弹窗
   143→   143→  const [dialogOpen, setDialogOpen] = useState(false);
   144→   144→  const [unlockEmail, setUnlockEmail] = useState("");
   145→   145→  const [unlockSubmitting, setUnlockSubmitting] = useState(false);
   146→   146→  const pendingRunRef = useRef<Record<string, string> | null>(null);
   147→   147→  const autoStartedRef = useRef(false);
   148→   148→
   149→   149→  /* ---------- SSE 流式执行 ---------- */
   150→   150→  const runInput = useCallback(
   151→   151→    async (payload: Record<string, string>) => {
   152→   152→      setStreaming(true);
   153→   153→      setError(null);
   154→   154→      setInlineError(null);
   155→   155→      setOutput("");
   156→   156→      setRunId((n) => n + 1);
   157→   157→      setStageLines([]);
   158→   158→      setProgress(5);
   159→   159→      setScanUrl(tool.isAnalyzer ? (payload.url ?? "").trim() || null : null);
   160→   160→      track(tool.isAnalyzer ? "scan_start" : "tool_start", { tool: tool.slug });
   161→   161→      let prog = 5;
   162→   162→      try {
   163→   163→        const res = await fetch(tool.endpoint, {
   164→   164→          method: "POST",
   165→   165→          headers: { "Content-Type": "application/json" },
   166→   166→          body: JSON.stringify(
   167→   167→            tool.isAnalyzer ? { url: payload.url } : { input: payload }
   168→   168→          ),
   169→   169→        });
   170→   170→
   171→   171→        if (!res.ok) {
   172→   172→          let msg = r.runFailed;
   173→   173→          try {
   174→   174→            const j = await res.json();
   175→   175→            if (j?.error) msg = j.error;
   176→   176→          } catch {
   177→   177→            /* 非 JSON 错误体 */
   178→   178→          }
   179→   179→          throw new Error(msg);
   180→   180→        }
   181→   181→
   182→   182→        const reader = res.body?.getReader();
   183→   183→        if (!reader) throw new Error(r.streamUnsupported);
   184→   184→
   185→   185→        const decoder = new TextDecoder();
   186→   186→        let buf = "";
   187→   187→        let acc = "";
   188→   188→
   189→   189→        for (;;) {
   190→   190→          const { done, value } = await reader.read();
   191→   191→          if (done) break;
   192→   192→          buf += decoder.decode(value, { stream: true });
   193→   193→          const parts = buf.split("\n\n");
   194→   194→          buf = parts.pop() ?? "";
   195→   195→          for (const part of parts) {
   196→   196→            const line = part.trim();
   197→   197→            if (!line.startsWith("data:")) continue;
   198→   198→            let evt: {
   199→   199→              type?: string;
   200→   200→              text?: string;
   201→   201→              error?: string;
   202→   202→              message?: string;
   203→   203→              progress?: number;
   204→   204→            };
   205→   205→            try {
   206→   206→              evt = JSON.parse(line.slice(5).trim());
   207→   207→            } catch {
   208→   208→              continue; // 半包 JSON，跳过
   209→   209→            }
   210→   210→            if (evt.type === "stage" && evt.message) {
   211→   211→              // 扫描阶段事件（/api/analyze）：驱动 ScanMonitor 日志与进度
   212→   212→              prog = typeof evt.progress === "number" ? evt.progress : prog;
   213→   213→              setProgress(prog);
   214→   214→              setStageLines((lines) => [...lines, evt.message as string]);
   215→   215→            } else if (evt.type === "delta" && evt.text) {
   216→   216→              if (prog < 72) {
   217→   217→                prog = 72; // 正文开始流出 → 进度推进到推理阶段
   218→   218→                setProgress(prog);
   219→   219→              }
   220→   220→              acc += evt.text;
   221→   221→              setOutput(acc);
   222→   222→            } else if (evt.type === "error") {
   223→   223→              throw new Error(evt.error || r.interrupted);
   224→   224→            }
   225→   225→          }
   226→   226→        }
   227→   227→
   228→   228→        if (acc.trim().length === 0) {
   229→   229→          setOutput(null);
   230→   230→          throw new Error(r.incomplete);
   231→   231→        }
   232→   232→        setProgress(100);
   233→   233→        // 完成记录：埋点 + 本地历史（历史容量与输出截断防止 localStorage 膨胀）
   234→   234→        track(tool.isAnalyzer ? "scan_complete" : "tool_complete", {
   235→   235→          tool: tool.slug,
   236→   236→          meta: `${acc.length} 字`,
   237→   237→        });
   238→   238→        setHistory(
   239→   239→          appendHistory(tool.slug, {
   240→   240→            ts: Date.now(),
   241→   241→            values: payload,
   242→   242→            output: acc.slice(0, HISTORY_OUTPUT_CAP),
   243→   243→            chars: acc.length,
   244→   244→          })
   245→   245→        );
   246→   246→      } catch (err) {
   247→   247→        const msg =
   248→   248→          err instanceof Error ? err.message : r.runFailed;
   249→   249→        track(tool.isAnalyzer ? "scan_fail" : "tool_error", {
   250→   250→          tool: tool.slug,
   251→   251→          meta: msg.slice(0, 80),
   252→   252→        });
   253→   253→        setError(msg);
   254→   254→        setOutput((prev) =>
   255→   255→          prev !== null && prev.trim().length >= 100 ? prev : null
   256→   256→        );
   257→   257→      } finally {
   258→   258→        setStreaming(false);
   259→   259→      }
   260→   260→    },
   261→   261→    [tool.endpoint, tool.isAnalyzer, r]
   262→   262→  );
   263→   263→
   264→   264→  /* ---------- 门控执行：localStorage 记账 ---------- */
   265→   265→  const runWithGate = useCallback(
   266→   266→    (payload: Record<string, string>) => {
   267→   267→      const unlockedMap = readJsonMap<boolean>(UNLOCKED_KEY);
   268→   268→      if (unlockedMap[tool.slug]) {
   269→   269→        setUnlocked(true);
   270→   270→        void runInput(payload);
   271→   271→        return;
   272→   272→      }
   273→   273→      const usageMap = readJsonMap<number>(USAGE_KEY);
   274→   274→      setUsage(usageMap);
   275→   275→      setUnlocked(false);
   276→   276→      const count = usageMap[tool.slug] ?? 0;
   277→   277→      if (count >= FREE_RUNS_PER_TOOL) {
   278→   278→        // 第 3 次：弹解锁弹窗，解锁成功后自动续跑
   279→   279→        track("unlock_modal", { tool: tool.slug });
   280→   280→        pendingRunRef.current = payload;
   281→   281→        setDialogOpen(true);
   282→   282→        return;
   283→   283→      }
   284→   284→      const next: UsageMap = { ...usageMap, [tool.slug]: count + 1 };
   285→   285→      writeJsonMap(USAGE_KEY, next);
   286→   286→      setUsage(next);
   287→   287→      void runInput(payload);
   288→   288→    },
   289→   289→    [tool.slug, runInput]
   290→   290→  );
   291→   291→
   292→   292→  /* ---------- 挂载：读 localStorage + ?url= 预填/自动跑 ---------- */
   293→   293→  useEffect(() => {
   294→   294→    if (autoStartedRef.current) return;
   295→   295→    autoStartedRef.current = true;
   296→   296→
   297→   297→    setUsage(readJsonMap<number>(USAGE_KEY));
   298→   298→    setUnlocked(!!readJsonMap<boolean>(UNLOCKED_KEY)[tool.slug]);
   299→   299→    setHistory(readHistory(tool.slug));
   300→   300→
   301→   301→    const url = new URLSearchParams(window.location.search).get("url");
   302→   302→    if (!tool.isAnalyzer || !url) return;
   303→   303→
   304→   304→    const norm = normalizeUrl(url);
   305→   305→    if (isValidUrl(norm)) {
   306→   306→      setValues((v) => ({ ...v, url: norm }));
   307→   307→      runWithGate({ url: norm });
   308→   308→    } else {
   309→   309→      setValues((v) => ({ ...v, url }));
   310→   310→    }
   311→   311→  }, [tool.slug, tool.isAnalyzer, runWithGate]);
   312→   312→
   313→   313→  /* ---------- 提交 ---------- */
   314→   314→  const startRun = () => {
   315→   315→    if (streaming) return;
   316→   316→    const missing = tool.fields.filter(
   317→   317→      (f) => !f.optional && !(values[f.id] ?? "").trim()
   318→   318→    );
   319→   319→    if (missing.length > 0) {
   320→   320→      setInlineError(r.fillRequired);
   321→   321→      return;
   322→   322→    }
   323→   323→
   324→   324→    let payload: Record<string, string> = Object.fromEntries(
   325→   325→      tool.fields.map((f) => [f.id, (values[f.id] ?? "").trim()])
   326→   326→    );
   327→   327→    if (tool.isAnalyzer) {
   328→   328→      const norm = normalizeUrl(payload.url);
   329→   329→      if (!isValidUrl(norm)) {
   330→   330→        setInlineError(r.invalidUrl);
   331→   331→        return;
   332→   332→      }
   333→   333→      payload = { url: norm };
   334→   334→      setValues((v) => ({ ...v, url: norm }));
   335→   335→    }
   336→   336→    runWithGate(payload);
   337→   337→  };
   338→   338→
   339→   339→  /* ---------- 解锁提交 ---------- */
   340→   340→  const handleUnlockSubmit = async (e: React.FormEvent) => {
   341→   341→    e.preventDefault();
   342→   342→    if (unlockSubmitting) return;
   343→   343→    const email = unlockEmail.trim();
   344→   344→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   345→   345→      toast({
   346→   346→        title: r.emailInvalid,
   347→   347→        description: r.emailInvalidDesc,
   348→   348→        variant: "destructive",
   349→   349→      });
   350→   350→      return;
   351→   351→    }
   352→   352→
   353→   353→    setUnlockSubmitting(true);
   354→   354→    try {
   355→   355→      // 首选公共线索接口（source: "tools"）；若枚举校验拒绝 → 回落专用解锁接口
   356→   356→      const leadBody = {
   357→   357→        name: r.unlockLeadName,
   358→   358→        contact: email,
   359→   359→        source: "tools",
   360→   360→        message: tpl(r.unlockLeadMsg, { slug: tool.slug }),
   361→   361→      };
   362→   362→      let ok = false;
   363→   363→      try {
   364→   364→        const res = await fetch("/api/leads", {
   365→   365→          method: "POST",
   366→   366→          headers: { "Content-Type": "application/json" },
   367→   367→          body: JSON.stringify(leadBody),
   368→   368→        });
   369→   369→        ok = res.ok;
   370→   370→      } catch {
   371→   371→        /* 网络错误 → 走回落 */
   372→   372→      }
   373→   373→      if (!ok) {
   374→   374→        const res2 = await fetch("/api/tools/unlock", {
   375→   375→          method: "POST",
   376→   376→          headers: { "Content-Type": "application/json" },
   377→   377→          body: JSON.stringify({ email, slug: tool.slug }),
   378→   378→        });
   379→   379→        if (!res2.ok) {
   380→   380→          let msg = r.submitFail;
   381→   381→          try {
   382→   382→            const j = await res2.json();
   383→   383→            if (j?.error) msg = j.error;
   384→   384→          } catch {
   385→   385→            /* 非 JSON 错误体 */
   386→   386→          }
   387→   387→          throw new Error(msg);
   388→   388→        }
   389→   389→      }
   390→   390→
   391→   391→      const map = readJsonMap<boolean>(UNLOCKED_KEY);
   392→   392→      map[tool.slug] = true;
   393→   393→      writeJsonMap(UNLOCKED_KEY, map);
   394→   394→      setUnlocked(true);
   395→   395→      track("unlock_success", { tool: tool.slug });
   396→   396→      toast({
   397→   397→        title: r.toastUnlocked,
   398→   398→        description: r.toastUnlockedDesc,
   399→   399→      });
   400→   400→      setDialogOpen(false);
   401→   401→      setUnlockEmail("");
   402→   402→      const pending = pendingRunRef.current;
   403→   403→      pendingRunRef.current = null;
   404→   404→      if (pending) void runInput(pending);
   405→   405→    } catch (err) {
   406→   406→      toast({
   407→   407→        title: r.unlockFail,
   408→   408→        description:
   409→   409→          err instanceof Error
   410→   410→            ? err.message
   411→   411→            : r.unlockFailHint,
   412→   412→        variant: "destructive",
   413→   413→      });
   414→   414→    } finally {
   415→   415→      setUnlockSubmitting(false);
   416→   416→    }
   417→   417→  };
   418→   418→
   419→   419→  /* ---------- 复制结果 ---------- */
   420→   420→  const copyOutput = async () => {
   421→   421→    if (!output) return;
   422→   422→    try {
   423→   423→      await navigator.clipboard.writeText(output);
   424→   424→      setCopied(true);
   425→   425→      toast({
   426→   426→        title: tool.isAnalyzer ? r.copiedAnalysisToast : r.copiedResultToast,
   427→   427→        description: r.copyHint,
   428→   428→      });
   429→   429→      setTimeout(() => setCopied(false), 2000);
   430→   430→    } catch {
   431→   431→      toast({
   432→   432→        title: r.copyFail,
   433→   433→        description: r.copyFailHint,
   434→   434→        variant: "destructive",
   435→   435→      });
   436→   436→    }
   437→   437→  };
   438→   438→
   439→   439→  const freeCount = usage?.[tool.slug] ?? 0;
   440→   440→  const remaining = Math.max(0, FREE_RUNS_PER_TOOL - freeCount);
   441→   441→  const outputLabel = tool.isAnalyzer ? r.copyAnalysis : r.copyResult;
   442→   442→
   443→   443→  return (
   444→   444→    <div>
   445→   445→      {/* 输入表单 */}
   446→   446→      <form
   447→   447→        onSubmit={(e) => {
   448→   448→          e.preventDefault();
   449→   449→          startRun();
   450→   450→        }}
   451→   451→        className="rounded-2xl border border-[#2A2A2A] bg-bhai-card p-6 sm:p-8 space-y-5"
   452→   452→      >
   453→   453→        {lt.fields.map((field) => (
   454→   454→          <div key={field.id}>
   455→   455→            <label
   456→   456→              htmlFor={`tool-${field.id}`}
   457→   457→              className="block text-sm font-medium text-foreground mb-2"
