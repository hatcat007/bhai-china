    31→const USAGE_KEY = "bhai_tool_usage";
    32→const UNLOCKED_KEY = "bhai_tool_unlocked";
    33→const HISTORY_KEY = "bhai_tool_history";
    34→const HISTORY_CAP = 5; // 每工具保留最近 5 次生成
    35→const HISTORY_OUTPUT_CAP = 8000; // 单条输出最多入库字数（控 localStorage 体积）
    36→
    37→type UsageMap = Record<string, number>;
    38→type UnlockedMap = Record<string, boolean>;

    40→type HistoryEntry = {
    41→  ts: number; // 完成时间戳
    42→  values: Record<string, string>; // 当时的表单输入
    43→  output: string; // 完整输出（截断至 8000 字）
    44→  chars: number; // 原始字数
    45→};

    47→function readJsonMap<T>(key: string): Record<string, T> {
    48→  if (typeof window === "undefined") return {};
    49→  try {
    50→    const raw = window.localStorage.getItem(key);
    51→    if (!raw) return {};
    52→    const parsed: unknown = JSON.parse(raw);
    53→    return parsed && typeof parsed === "object"
    54→      ? (parsed as Record<string, T>)
    55→      : {};
    56→  } catch {
    57→    return {};
    58→  }
    59→}
    60→
    61→function writeJsonMap(key: string, map: Record<string, unknown>): void {
    62→  try {
    63→    window.localStorage.setItem(key, JSON.stringify(map));
    64→  } catch {
    65→    /* 私密模式等场景静默失败 */
    66→  }
    67→}
    68→
    69→function readHistory(slug: string): HistoryEntry[] {
    70→  const map = readJsonMap<HistoryEntry[]>(HISTORY_KEY);
    71→  const list = map[slug];
    72→  return Array.isArray(list) ? list : [];
    73→}
    74→
    75→function appendHistory(slug: string, entry: HistoryEntry): HistoryEntry[] {
    76→  const map = readJsonMap<HistoryEntry[]>(HISTORY_KEY);
    77→  const list = [entry, ...(map[slug] ?? [])].slice(0, HISTORY_CAP);
    78→  map[slug] = list;
    79→  writeJsonMap(HISTORY_KEY, map);
    80→  return list;
    81→}
    82→
    83→function clearHistory(slug: string): HistoryEntry[] {
    84→  const map = readJsonMap<HistoryEntry[]>(HISTORY_KEY);
    85→  delete map[slug];
    86→  writeJsonMap(HISTORY_KEY, map);
    87→  return [];
    88→}

    90→function fmtTime(ts: number): string {
    91→  const d = new Date(ts);
    92→  const p = (n: number) => String(n).padStart(2, "0");
    93→  return `${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}`;
    94→}

    96→function normalizeUrl(raw: string): string {
    97→  const s = raw.trim();
    98→  if (!s) return s;
    99→  return /^https?:\/\//i.test(s) ? s : `https://${s}`;
   100→}
   101→
   102→function isValidUrl(raw: string): boolean {
   103→  try {
   104→    const u = new URL(raw);
   105→    return (u.protocol === "http:" || u.protocol === "https:") && !!u.hostname;
   106→  } catch {
   107→    return false;
   108→  }
   109→}

   113→export function ToolRunner({ tool }: { tool: AITool }) {
   114→  const { toast } = useToast();
   115→  const Icon = TOOL_ICONS[tool.icon] ?? TOOL_ICONS.ScanSearch;
   116→
   117→  const [values, setValues] = useState<Record<string, string>>(() =>
   118→    Object.fromEntries(tool.fields.map((f) => [f.id, ""]))
   119→  );
   120→  const [usage, setUsage] = useState<UsageMap | null>(null); // null = localStorage 未读
   121→  const [unlocked, setUnlocked] = useState(false);
   122→  const [output, setOutput] = useState<string | null>(null);
   123→  const [streaming, setStreaming] = useState(false);
   124→  const [error, setError] = useState<string | null>(null);
   125→  const [inlineError, setInlineError] = useState<string | null>(null);
   126→  const [copied, setCopied] = useState(false);
   127→
   128→  // 扫描监视器状态（仅分析器使用）：阶段日志 / 进度 / 轮次号（重挂载用）
   129→  const [stageLines, setStageLines] = useState<string[]>([]);
   130→  const [progress, setProgress] = useState(0);
   131→  const [runId, setRunId] = useState(0);
   132→  // 提交时冻结的目标 URL：避免输入过程中实时触发截图预加载/回退链空转
   133→  const [scanUrl, setScanUrl] = useState<string | null>(null);
   134→  // 本工具最近生成记录（localStorage，最多 5 条）
   135→  const [history, setHistory] = useState<HistoryEntry[]>([]);
   136→  const [historyOpen, setHistoryOpen] = useState(false);
   137→
   138→  // 解锁弹窗
   139→  const [dialogOpen, setDialogOpen] = useState(false);
   140→  const [unlockEmail, setUnlockEmail] = useState("");
   141→  const [unlockSubmitting, setUnlockSubmitting] = useState(false);
   142→  const pendingRunRef = useRef<Record<string, string> | null>(null);
   143→  const autoStartedRef = useRef(false);
   144→
   145→  /* ---------- SSE 流式执行 ---------- */
   146→  const runInput = useCallback(
   147→    async (payload: Record<string, string>) => {
   148→      setStreaming(true);
   149→      setError(null);
   150→      setInlineError(null);
   151→      setOutput("");
   152→      setRunId((n) => n + 1);
   153→      setStageLines([]);
   154→      setProgress(5);
   155→      setScanUrl(tool.isAnalyzer ? (payload.url ?? "").trim() || null : null);
   156→      track(tool.isAnalyzer ? "scan_start" : "tool_start", { tool: tool.slug });
   157→      let prog = 5;
   158→      try {
   159→        const res = await fetch(tool.endpoint, {
   160→          method: "POST",
   161→          headers: { "Content-Type": "application/json" },
   162→          body: JSON.stringify(
   163→            tool.isAnalyzer ? { url: payload.url } : { input: payload }
   164→          ),
   165→        });
   166→
   167→        if (!res.ok) {
   168→          let msg = "生成失败，请稍后重试";
   169→          try {
   170→            const j = await res.json();
   171→            if (j?.error) msg = j.error;
   172→          } catch {
   173→            /* 非 JSON 错误体 */
   174→          }
   175→          throw new Error(msg);
   176→        }
   177→
   178→        const reader = res.body?.getReader();
   179→        if (!reader) throw new Error("浏览器不支持流式读取");
   180→
   181→        const decoder = new TextDecoder();
   182→        let buf = "";
   183→        let acc = "";
   184→
   185→        for (;;) {
   186→          const { done, value } = await reader.read();
   187→          if (done) break;
   188→          buf += decoder.decode(value, { stream: true });
   189→          const parts = buf.split("\n\n");
   190→          buf = parts.pop() ?? "";
   191→          for (const part of parts) {
   192→            const line = part.trim();
   193→            if (!line.startsWith("data:")) continue;
   194→            let evt: {
   195→              type?: string;
   196→              text?: string;
   197→              error?: string;
   198→              message?: string;
   199→              progress?: number;
   200→            };
   201→            try {
   202→              evt = JSON.parse(line.slice(5).trim());
   203→            } catch {
   204→              continue; // 半包 JSON，跳过
   205→            }
   206→            if (evt.type === "stage" && evt.message) {
   207→              // 扫描阶段事件（/api/analyze）：驱动 ScanMonitor 日志与进度
   208→              prog = typeof evt.progress === "number" ? evt.progress : prog;
   209→              setProgress(prog);
   210→              setStageLines((lines) => [...lines, evt.message as string]);
   211→            } else if (evt.type === "delta" && evt.text) {
   212→              if (prog < 72) {
   213→                prog = 72; // 正文开始流出 → 进度推进到推理阶段
   214→                setProgress(prog);
   215→              }
   216→              acc += evt.text;
   217→              setOutput(acc);
   218→            } else if (evt.type === "error") {
   219→              throw new Error(evt.error || "生成中断");
   220→            }
   221→          }
   222→        }
   223→
   224→        if (acc.trim().length === 0) {
   225→          setOutput(null);
   226→          throw new Error("生成内容不完整，请重试");
   227→        }
   228→        setProgress(100);
   229→        // 完成记录：埋点 + 本地历史（历史容量与输出截断防止 localStorage 膨胀）
   230→        track(tool.isAnalyzer ? "scan_complete" : "tool_complete", {
   231→          tool: tool.slug,
   232→          meta: `${acc.length} 字`,
   233→        });
   234→        setHistory(
   235→          appendHistory(tool.slug, {
   236→            ts: Date.now(),
   237→            values: payload,
   238→            output: acc.slice(0, HISTORY_OUTPUT_CAP),
   239→            chars: acc.length,
   240→          })
   241→        );
   242→      } catch (err) {
   243→        const msg =
   244→          err instanceof Error ? err.message : "生成失败，请稍后重试";
   245→        track(tool.isAnalyzer ? "scan_fail" : "tool_error", {
   246→          tool: tool.slug,
   247→          meta: msg.slice(0, 80),
   248→        });
   249→        setError(msg);
   250→        setOutput((prev) =>
   251→          prev !== null && prev.trim().length >= 100 ? prev : null
   252→        );
   253→      } finally {
   254→        setStreaming(false);
   255→      }
   256→    },
   257→    [tool.endpoint, tool.isAnalyzer]
   258→  );
   259→
   260→  /* ---------- 门控执行：localStorage 记账 ---------- */
   261→  const runWithGate = useCallback(
   262→    (payload: Record<string, string>) => {
   263→      const unlockedMap = readJsonMap<boolean>(UNLOCKED_KEY);
   264→      if (unlockedMap[tool.slug]) {
   265→        setUnlocked(true);
   266→        void runInput(payload);
   267→        return;
   268→      }
   269→      const usageMap = readJsonMap<number>(USAGE_KEY);
   270→      setUsage(usageMap);
   271→      setUnlocked(false);
   272→      const count = usageMap[tool.slug] ?? 0;
   273→      if (count >= FREE_RUNS_PER_TOOL) {
   274→        // 第 3 次：弹解锁弹窗，解锁成功后自动续跑
   275→        track("unlock_modal", { tool: tool.slug });
   276→        pendingRunRef.current = payload;
   277→        setDialogOpen(true);
   278→        return;
   279→      }
   280→      const next: UsageMap = { ...usageMap, [tool.slug]: count + 1 };
   281→      writeJsonMap(USAGE_KEY, next);
   282→      setUsage(next);
   283→      void runInput(payload);
   284→    },
   285→    [tool.slug, runInput]
   286→  );
   287→
   288→  /* ---------- 挂载：读 localStorage + ?url= 预填/自动跑 ---------- */
   289→  useEffect(() => {
   290→    if (autoStartedRef.current) return;
   291→    autoStartedRef.current = true;
   292→
   293→    setUsage(readJsonMap<number>(USAGE_KEY));
   294→    setUnlocked(!!readJsonMap<boolean>(UNLOCKED_KEY)[tool.slug]);
   295→    setHistory(readHistory(tool.slug));
   296→
   297→    const url = new URLSearchParams(window.location.search).get("url");
   298→    if (!tool.isAnalyzer || !url) return;
   299→
   300→    const norm = normalizeUrl(url);
   301→    if (isValidUrl(norm)) {
   302→      setValues((v) => ({ ...v, url: norm }));
   303→      runWithGate({ url: norm });
   304→    } else {
   305→      setValues((v) => ({ ...v, url }));
   306→      setInlineError(INVALID_URL_MSG);
   307→    }
   308→  }, [tool.slug, tool.isAnalyzer, runWithGate]);
   309→
   310→  /* ---------- 提交 ---------- */
   311→  const startRun = () => {
   312→    if (streaming) return;
   313→    const missing = tool.fields.filter(
   314→      (f) => !f.optional && !(values[f.id] ?? "").trim()
   315→    );
   316→    if (missing.length > 0) {
   317→      setInlineError("请把必填项填完整再生成");
   318→      return;
   319→    }
   320→
   321→    let payload: Record<string, string> = Object.fromEntries(
   322→      tool.fields.map((f) => [f.id, (values[f.id] ?? "").trim()])
   323→    );
   324→    if (tool.isAnalyzer) {
   325→      const norm = normalizeUrl(payload.url);
   326→      if (!isValidUrl(norm)) {
   327→        setInlineError(INVALID_URL_MSG);
   328→        return;
   329→      }
   330→      payload = { url: norm };
   331→      setValues((v) => ({ ...v, url: norm }));
   332→    }
   333→    runWithGate(payload);
   334→  };
   335→
   336→  /* ---------- 解锁提交 ---------- */
   337→  const handleUnlockSubmit = async (e: React.FormEvent) => {
   338→    e.preventDefault();
   339→    if (unlockSubmitting) return;
   340→    const email = unlockEmail.trim();
   341→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   342→      toast({
   343→        title: "邮箱格式不对",
   344→        description: "检查一下再提交，或者微信联系 busterl1",
   345→        variant: "destructive",
   346→      });
   347→      return;
   348→    }
   349→
   350→    setUnlockSubmitting(true);
   351→    try {
   352→      // 首选公共线索接口（source: "tools"）；若枚举校验拒绝 → 回落专用解锁接口
   353→      const leadBody = {
   354→        name: "工具用户",
   355→        contact: email,
   356→        source: "tools",
   357→        message: `解锁工具: ${tool.slug}`,
   358→      };
   359→      let ok = false;
   360→      try {
   361→        const res = await fetch("/api/leads", {
   362→          method: "POST",
   363→          headers: { "Content-Type": "application/json" },
   364→          body: JSON.stringify(leadBody),
   365→        });
   366→        ok = res.ok;
   367→      } catch {
   368→        /* 网络错误 → 走回落 */
   369→      }
   370→      if (!ok) {
   371→        const res2 = await fetch("/api/tools/unlock", {
   372→          method: "POST",
   373→          headers: { "Content-Type": "application/json" },
   374→          body: JSON.stringify({ email, slug: tool.slug }),
   375→        });
   376→        if (!res2.ok) {
   377→          let msg = "提交失败，请稍后再试";
   378→          try {
   379→            const j = await res2.json();
   380→            if (j?.error) msg = j.error;
   381→          } catch {
   382→            /* 非 JSON 错误体 */
   383→          }
   384→          throw new Error(msg);
   385→        }
   386→      }
   387→
   388→      const map = readJsonMap<boolean>(UNLOCKED_KEY);
   389→      map[tool.slug] = true;
   390→      writeJsonMap(UNLOCKED_KEY, map);
   391→      setUnlocked(true);
   392→      track("unlock_success", { tool: tool.slug });
   393→      toast({
   394→        title: "已解锁",
   395→        description: "无限次已开启，《CEO AI 清单》本周发出。",
   396→      });
   397→      setDialogOpen(false);
   398→      setUnlockEmail("");
   399→      const pending = pendingRunRef.current;
   400→      pendingRunRef.current = null;
   401→      if (pending) void runInput(pending);
   402→    } catch (err) {
   403→      toast({
   404→        title: "解锁失败",
   405→        description:
   406→          err instanceof Error
   407→            ? err.message
   408→            : "请稍后再试或微信联系 busterl1",
   409→        variant: "destructive",
   410→      });
   411→    } finally {
   412→      setUnlockSubmitting(false);
   413→    }
   414→  };
   415→
   416→  /* ---------- 复制结果 ---------- */
   417→  const copyOutput = async () => {
   418→    if (!output) return;
   419→    try {
   420→      await navigator.clipboard.writeText(output);
   421→      setCopied(true);
   422→      toast({
   423→        title: tool.isAnalyzer ? "分析结果已复制" : "结果已复制",
   424→        description: "粘贴到微信或文档里直接用",
   425→      });
   426→      setTimeout(() => setCopied(false), 2000);
   427→    } catch {
   428→      toast({
   429→        title: "复制失败",
   430→        description: "请手动选中文字复制",
   431→        variant: "destructive",
   432→      });
   433→    }
   434→  };
   435→
   436→  const freeCount = usage?.[tool.slug] ?? 0;
   437→  const remaining = Math.max(0, FREE_RUNS_PER_TOOL - freeCount);
   438→  const outputLabel = tool.isAnalyzer ? "复制分析结果" : "复制结果";
   439→
   440→  return (
   441→    <div>
   442→      {/* 输入表单 */}
   443→      <form
   444→        onSubmit={(e) => {
   445→          e.preventDefault();
   446→          startRun();
   447→        }}
   448→        className="rounded-2xl border border-[#2A2A2A] bg-bhai-card p-6 sm:p-8 space-y-5"
   449→      >
   450→        {tool.fields.map((field) => (
   451→          <div key={field.id}>
   452→            <label
   453→              htmlFor={`tool-${field.id}`}
   454→              className="block text-sm font-medium text-foreground mb-2"
   455→            >
   456→              {field.label}
   457→              {field.optional && (
   458→                <span className="ml-1.5 font-normal text-bhai-dim">（可选）</span>
   459→              )}
   460→            </label>
   461→            <input
   462→              id={`tool-${field.id}`}
   463→              name={field.id}
   464→              type={field.type === "url" ? "url" : "text"}
   465→              placeholder={field.placeholder}
   466→              maxLength={field.maxLength}
   467→              value={values[field.id] ?? ""}
   468→              onChange={(e) =>
   469→                setValues((v) => ({ ...v, [field.id]: e.target.value }))
   470→              }
   471→              disabled={streaming}
   472→              autoComplete={field.type === "url" ? "url" : "off"}
   473→              className="w-full rounded-lg border border-[#2A2A2A] bg-bhai-bg px-4 py-3 text-base text-foreground placeholder:text-bhai-dim input-focus disabled:opacity-50"
   474→            />
   475→          </div>
   476→        ))}
   477→
   478→        {inlineError && (
   479→          <p role="alert" className="text-xs text-red-400">
   480→            {inlineError}
   481→          </p>
   482→        )}
   483→
   484→        <div className="pt-1 space-y-3">
   485→          <button
   486→            type="submit"
   487→            disabled={streaming}
   488→            className="cta-primary w-full rounded-md px-6 py-3 text-sm font-medium text-white inline-flex items-center justify-center gap-2 disabled:opacity-60 disabled:cursor-wait"
   489→          >
   490→            {streaming ? (
   491→              <>
   492→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   493→                {tool.isAnalyzer ? "正在扫描…" : "生成中…"}
   494→              </>
   495→            ) : (
   496→              <>
   497→                <Icon className="h-4 w-4" aria-hidden="true" />
   498→                {tool.ctaText}
   499→              </>
   500→            )}
   501→          </button>
   502→
   503→          <div className="flex flex-wrap items-center justify-between gap-2">
   504→            <p className="font-mono text-[10px] text-bhai-dim tracking-wider">
   505→              {tool.outputHint}
   506→            </p>
   507→            {unlocked ? (
   508→              <span className="inline-flex items-center gap-1 rounded-full border border-bhai-red/40 bg-bhai-red/10 px-2.5 py-0.5 font-mono text-[10px] text-bhai-red shrink-0">
   509→                已解锁 ∞
   510→              </span>
   511→            ) : (
   512→              usage !== null && (
   513→                <span className="font-mono text-[10px] text-bhai-dim shrink-0">
   514→                  {remaining > 0
   515→                    ? `还剩 ${remaining} 次免费`
   516→                    : "免费额度已用完 · 留邮箱解锁无限次"}
   517→                </span>
   518→              )
   519→            )}
   520→          </div>
   521→        </div>
   522→      </form>
   523→
   524→      {/* 扫描监视器（仅网站分析器）：目标站截图 + 向下滚动 + 扫描线动画 + 阶段日志 */}
   525→      {tool.isAnalyzer && (
   526→        <div className="mt-6">
   527→          <ScanMonitor
   528→            key={runId}
   529→            url={scanUrl}
   530→            active={streaming}
   531→            done={!streaming && output !== null && !error}
   532→            failed={!streaming && error !== null}
   533→            stageLines={stageLines}
   534→            progress={progress}
   535→          />
   536→        </div>
   537→      )}
   538→
   539→      {/* 错误提示（保留部分输出时与面板并存） */}
